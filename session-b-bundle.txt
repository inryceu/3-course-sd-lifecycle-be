  async update(userId: string, cardId: string, input: UpdateCardInput): Promise<CardView> {
    const card = await this.findCard(cardId);
    await this.access.requireEditor(userId, card.boardId);

    if (input.title !== undefined) card.title = input.title;
    if (input.description !== undefined) card.description = input.description;
    if (input.deadline !== undefined) {
      const deadline = input.deadline ? new Date(input.deadline) : null;
      // A new deadline gets its own reminder.
      if (deadline?.getTime() !== card.deadline?.getTime()) card.reminderSentAt = null;
      card.deadline = deadline;
    }
    const saved = await this.cards.save(card);

    const view = toCardView(saved);
    await this.events.publish(
      createEvent('card.updated', {
        boardId: card.boardId,
        actorId: userId,
        origin: 'user',
        payload: toCardSnapshot(view),
      }),
    );
    return view;
  }

  async move(userId: string, cardId: string, input: MoveCardInput): Promise<CardView> {
    const card = await this.findCard(cardId);
    const { boardId } = card;
    await this.access.requireEditor(userId, boardId);

    const { view, fromColumnId } = await this.dataSource.transaction(async (manager) => {
      await this.lockBoard(manager, boardId);

      const current = await manager.findOne(CardEntity, {
        where: { id: cardId },
        relations: { labels: true },
      });
      const target = await manager.findOne(BoardColumnEntity, { where: { id: input.columnId } });
      if (!current) throw new NotFoundException('Card not found');
      if (!target) throw new NotFoundException('Target column not found');

      const from = { columnId: current.columnId, position: current.position };
      const sameColumn = from.columnId === target.id;
      const count = await manager.count(CardEntity, { where: { columnId: target.id } });
      const maxPosition = sameColumn ? count - 1 : count;
      if (input.position > maxPosition) {
        throw new BadRequestException(`Position must be between 0 and ${maxPosition}`);
      }

      try {
        current.moveTo({ id: target.id, boardId: target.boardId }, input.position);
      } catch (error) {
        if (error instanceof CrossBoardMoveError || error instanceof InvalidPositionError) {
          throw new BadRequestException(error.message);
        }
        throw error;
      }

      await this.shiftNeighbours(manager, from, { columnId: target.id, position: input.position });
      await manager.update(
        CardEntity,
        { id: cardId },
        { columnId: current.columnId, position: current.position },
      );
      const moved = await manager.findOneOrFail(CardEntity, {
        where: { id: cardId },
        relations: { labels: true },
      });
      return { view: toCardView(moved), fromColumnId: from.columnId };
    });

    await this.events.publish(
      createEvent('card.moved', {
        boardId,
        actorId: userId,
        origin: 'user',
        payload: {
          card: toCardSnapshot(view),
          fromColumnId,
          toColumnId: view.columnId,
          position: view.position,
        },
      }),
    );
    return view;
  }

  async remove(userId: string, cardId: string): Promise<void> {
    const card = await this.findCard(cardId);
    await this.access.requireEditor(userId, card.boardId);

    await this.dataSource.transaction(async (manager) => {
      await this.lockBoard(manager, card.boardId);
      const current = await manager.findOne(CardEntity, { where: { id: cardId } });
      if (!current) return;
      await manager.delete(CardEntity, { id: cardId });
      await manager
        .createQueryBuilder()
        .update(CardEntity)
        .set({ position: () => 'position - 1' })
        .where('column_id = :columnId AND position > :position', {
          columnId: current.columnId,
          position: current.position,
        })
        .execute();
    });

    await this.events.publish(
      createEvent('board.updated', {
        boardId: card.boardId,
        actorId: userId,
        origin: 'user',
        payload: { board: await this.boardSnapshot(card.boardId) },
      }),
    );
  }

  /**
   * Creates or updates the card of a Jira issue, matched by `jiraIssueKey`, and puts it in
   * `columnId` (appended at the end when it has to move). One transaction serialised per board.
   */
  async upsertFromExternal(input: ExternalCardUpsert): Promise<ExternalUpsertResult> {
    const { boardId, jiraIssueKey, columnId, title, description } = input;

    const result = await this.dataSource.transaction(async (manager) => {
      await this.lockBoard(manager, boardId);
      const target = await manager.findOne(BoardColumnEntity, { where: { id: columnId, boardId } });
      if (!target) {
        throw new BadRequestException('Column does not belong to this board');
      }
      const card = await manager.findOne(CardEntity, {
        where: { boardId, jiraIssueKey },
        relations: { labels: true },
      });

      if (!card) {
        const count = await manager.count(CardEntity, { where: { columnId } });
        const saved = await manager.save(
          manager.create(CardEntity, {
            boardId,
            columnId,
            title,
            description: description ?? null,
            jiraIssueKey,
            position: count,
            labels: [],
          }),
        );
        return { view: toCardView(saved), outcome: 'created' as const, fromColumnId: null };
      }

      const newDescription = description === undefined ? card.description : description;
      const textChanged = card.title !== title || card.description !== newDescription;
      const fromColumnId = card.columnId;
      if (!textChanged && fromColumnId === columnId) {
        return { view: toCardView(card), outcome: 'unchanged' as const, fromColumnId: null };
      }
      if (textChanged) {
        await manager.update(CardEntity, { id: card.id }, { title, description: newDescription });
      }
      if (fromColumnId !== columnId) {
        const count = await manager.count(CardEntity, { where: { columnId } });
        const from = { columnId: fromColumnId, position: card.position };
        card.moveTo({ id: target.id, boardId: target.boardId }, count);
        await this.shiftNeighbours(manager, from, { columnId, position: count });
        await manager.update(CardEntity, { id: card.id }, { columnId, position: count });
      }
      const saved = await manager.findOneOrFail(CardEntity, {
        where: { id: card.id },
        relations: { labels: true },
      });
      return {
        view: toCardView(saved),
        outcome: 'updated' as const,
        fromColumnId: fromColumnId !== columnId ? fromColumnId : null,
      };
    });

    const { view, outcome, fromColumnId } = result;
    if (outcome === 'created') {
      await this.events.publish(
        createEvent('card.created', { boardId, origin: 'jira', payload: toCardSnapshot(view) }),
      );
    } else if (outcome === 'updated') {
      await this.events.publish(
        fromColumnId
          ? createEvent('card.moved', {
              boardId,
              origin: 'jira',
              payload: {
                card: toCardSnapshot(view),
                fromColumnId,
                toColumnId: view.columnId,
                position: view.position,
              },
            })
          : createEvent('card.updated', { boardId, origin: 'jira', payload: toCardSnapshot(view) }),
      );
    }
    return { cardId: view.id, outcome };
  }

  async moveFromExternal(input: {
    boardId: string;
    jiraIssueKey: string;
    columnId: string;
    position?: number;
  }): Promise<'moved' | 'unchanged' | 'not-found'> {
    const { boardId, jiraIssueKey, columnId: targetColumnId, position: targetPosition } = input;

    const result = await this.dataSource.transaction(async (manager) => {
      await this.lockBoard(manager, boardId);
      const current = await manager.findOne(CardEntity, {
        where: { boardId, jiraIssueKey },
        relations: { labels: true },
      });
      if (!current) return 'not-found';

      const target = await manager.findOne(BoardColumnEntity, {
        where: { id: targetColumnId, boardId },
      });
      if (!target) return 'not-found';

      const from = { columnId: current.columnId, position: current.position };
      const sameColumn = from.columnId === target.id;
      // A status that maps to the card's current column did not move it; keep the user's order.
      if (sameColumn && targetPosition === undefined) return 'unchanged';
      const count = await manager.count(CardEntity, { where: { columnId: target.id } });
      const maxPosition = sameColumn ? count - 1 : count;
      const finalPosition =
        targetPosition === undefined
          ? maxPosition
          : Math.min(Math.max(0, targetPosition), maxPosition);

      if (from.columnId === target.id && from.position === finalPosition) {
        return 'unchanged';
      }

      current.moveTo({ id: target.id, boardId: target.boardId }, finalPosition);

      await this.shiftNeighbours(manager, from, { columnId: target.id, position: finalPosition });
      await manager.update(
        CardEntity,
        { id: current.id },
        { columnId: current.columnId, position: current.position },
      );

      const moved = await manager.findOneOrFail(CardEntity, {
        where: { id: current.id },
        relations: { labels: true },
      });
      return { view: toCardView(moved), fromColumnId: from.columnId };
    });

    if (result === 'not-found' || result === 'unchanged') {
      return result;
    }

    await this.events.publish(
      createEvent('card.moved', {
        boardId,
        origin: 'jira',
        payload: {
          card: toCardSnapshot(result.view),
          fromColumnId: result.fromColumnId,
          toColumnId: result.view.columnId,
          position: result.view.position,
        },
      }),
    );

    return 'moved';
  }

  private async shiftNeighbours(
    manager: EntityManager,
    from: { columnId: string; position: number },
    to: { columnId: string; position: number },
  ): Promise<void> {
    const shift = (columnId: string, delta: '+ 1' | '- 1', condition: string, params: object) =>
      manager
        .createQueryBuilder()
        .update(CardEntity)
        .set({ position: () => `position ${delta}` })
        .where(`column_id = :columnId AND ${condition}`, { columnId, ...params })
        .execute();

    if (from.columnId === to.columnId) {
      if (to.position > from.position) {
        await shift(from.columnId, '- 1', 'position > :from AND position <= :to', {
          from: from.position,
          to: to.position,
        });
      } else if (to.position < from.position) {
        await shift(from.columnId, '+ 1', 'position >= :to AND position < :from', {
          from: from.position,
          to: to.position,
        });
      }
      return;
    }
    await shift(from.columnId, '- 1', 'position > :from', { from: from.position });
    await shift(to.columnId, '+ 1', 'position >= :to', { to: to.position });
  }

  private async findColumn(columnId: string): Promise<BoardColumnEntity> {
    const column = await this.columns.findOne({ where: { id: columnId } });
    if (!column) {
      throw new NotFoundException('Column not found');
    }
    return column;
  }

  private async findCard(cardId: string): Promise<CardEntity> {
    const card = await this.cards.findOne({ where: { id: cardId }, relations: { labels: true } });
    if (!card) {
      throw new NotFoundException('Card not found');
    }
    return card;
  }

  private async resolveLabels(boardId: string, labelIds?: string[]): Promise<LabelEntity[]> {
    if (!labelIds?.length) return [];
    const labels = await this.dataSource
      .getRepository(LabelEntity)
      .find({ where: { id: In(labelIds), boardId } });
    if (labels.length !== new Set(labelIds).size) {
      throw new BadRequestException('Unknown label for this board');
    }
    return labels;
  }

  private async assertMember(boardId: string, userId: string): Promise<void> {
    const membership = await this.dataSource
      .getRepository(BoardMembershipEntity)
      .findOne({ where: { boardId, userId } });
    if (!membership) {
      throw new BadRequestException('The assignee is not a member of this board');
    }
  }

  private async boardSnapshot(boardId: string) {
    const board = await this.dataSource.getRepository(BoardEntity).findOneOrFail({
      where: { id: boardId },
    });
    return toBoardSnapshot(board);
  }

  private async lockBoard(manager: EntityManager, boardId: string): Promise<void> {
    const board = await manager
      .createQueryBuilder(BoardEntity, 'board')
      .setLock('pessimistic_write')
      .where('board.id = :boardId', { boardId })
      .getOne();
    if (!board) {
      throw new NotFoundException('Board not found');
    }
  }
}

type FailureReason =

/** Applies one user card move to Jira and records the attempt in `sync_logs`. */
@Injectable()
export class JiraStatusSyncService {
  private readonly logger = new Logger(JiraStatusSyncService.name);
  // ponytail: in-process only; T-35 per-card ordering (outbox) replaces it across processes.
  private readonly chains = new Map<string, Promise<void>>();

  constructor(
    private readonly statusMappings: JiraStatusMappingService,
    @Inject(JIRA_GATEWAY) private readonly gateway: JiraGateway,
    @InjectRepository(SyncLogEntity)
    private readonly logs: Repository<SyncLogEntity>,
    @InjectRepository(JiraIssueMappingEntity)
    private readonly links: Repository<JiraIssueMappingEntity>,
  ) {}

  async syncCardMove(event: DomainEventOf<'card.moved'>): Promise<void> {
    const issueKey = event.payload.card.jiraIssueKey;
    if (
      event.origin !== 'user' ||
      event.payload.fromColumnId === event.payload.toColumnId ||
      issueKey === null
    ) {
      return;
    }

    // Runs for one issue go one after another, so a later move reads Jira after the earlier
    // transition landed instead of racing it and leaving Jira in the older status.
    const key = `${event.boardId}:${issueKey}`;
    const next = (this.chains.get(key) ?? Promise.resolve()).then(() =>
      this.syncOne(event, issueKey),
    );
    const settled = next.catch((): void => undefined);
    this.chains.set(key, settled);
    void settled.then(() => {
      if (this.chains.get(key) === settled) this.chains.delete(key);
    });
    return next;
  }

  private async syncOne(event: DomainEventOf<'card.moved'>, issueKey: string): Promise<void> {
    const startedAt = this.startedAt(event.occurredAt);
    const deadline = startedAt + SYNC_BUDGET_MS;
    const result = await this.run(event, issueKey, deadline);
    const latencyMs = Math.max(0, Date.now() - startedAt);

    if (latencyMs > SYNC_BUDGET_MS) {
      this.logger.warn(
        `Jira status sync for ${issueKey} on board ${event.boardId} took ${latencyMs}ms`,
      );
    }
    if (result.status === SyncStatus.FAILED) {
      const what = result.payload.outcome === 'unknown' ? 'outcome unknown' : 'failed';
      this.logger.error(
        `Jira status sync ${what} for ${issueKey} on board ${event.boardId}: ${result.reason ?? 'sync failed'}`,
        result.errorMessage ?? undefined,
      );
    } else {
      this.logger.log(
        `Jira status sync for ${issueKey} on board ${event.boardId} completed in ${latencyMs}ms`,
      );
    }

    await this.logs.save({
      direction: SyncDirection.TO_JIRA,
      status: result.status,
      payload: { ...result.payload, latencyMs },
      errorMessage: result.errorMessage,
      mappingId: result.mappingId,
    });
  }

  private async run(
    event: DomainEventOf<'card.moved'>,
    issueKey: string,
    deadline: number,
  ): Promise<SyncLogResult> {
    const { card, fromColumnId, toColumnId } = event.payload;
    let mappingId: string | null = null;

    try {
      mappingId = await this.findMappingId(event.boardId, issueKey);
      const linked = await this.resolveLinkedBoard(event.boardId);
      if (!ISSUE_KEY_PATTERN.test(issueKey)) {
        throw new SyncFailure('invalid issue key', `"${issueKey}" is not a Jira issue key`);
      }

      const issue = await this.getIssue(event.boardId, issueKey, deadline);
      const [current] = await this.statusMappings.resolveMappings(linked, [issue.status]);
      if (current.columnId === toColumnId) {
        return {
          status: SyncStatus.SUCCESS,
          payload: {
            cardId: card.id,
            issueKey,
            fromColumnId,
            toColumnId,
            outcome: 'no-op',
            targetStatusId: issue.status.id,
            targetStatusName: issue.status.name,
          },
          errorMessage: null,
          mappingId,
        };
      }

      const transitions = await this.listTransitions(event.boardId, issueKey, deadline);
      const targets = this.uniqueTargets(transitions);
      const targetMappings = await this.statusMappings.resolveMappings(linked, targets);
      const mappingByStatusId = new Map(
        targetMappings.map((mapping) => [mapping.jiraStatusId, mapping]),
      );
      const candidates = transitions
        .filter((transition) => transition.isAvailable)
        .map((transition) => {
          const mapping = mappingByStatusId.get(transition.to.id);
          return mapping === undefined ? null : { transition, mapping };
        })
        .filter(
          (candidate): candidate is TransitionCandidate =>
            candidate !== null && candidate.mapping.columnId === toColumnId,
        )
        .sort((left, right) => this.compareCandidates(left, right));

      if (candidates.length === 0) {
        throw new SyncFailure(
          'no valid transition',
          `No available transition targets column ${toColumnId}`,
        );
      }

      const chosen = candidates[0];
      try {
        await this.doTransition(event.boardId, issueKey, chosen.transition.id, deadline);
      } catch (error) {
        // The POST was sent and aborted: Jira may have applied it.
        if (error instanceof JiraTimeoutError) {
          throw new SyncFailure('deadline exceeded', error.message, {
            outcome: 'unknown',
            transitionId: chosen.transition.id,
          });
        }
        throw error;
      }
      return {
        status: SyncStatus.SUCCESS,
        payload: {
          cardId: card.id,
          issueKey,
          fromColumnId,
          toColumnId,
          outcome: 'transitioned',
          targetStatusId: chosen.transition.to.id,
          targetStatusName: chosen.transition.to.name,
          transitionId: chosen.transition.id,
        },
        errorMessage: null,
        mappingId,
      };
    } catch (error) {
      const failure = this.failureFrom(error);
      return {
        status: SyncStatus.FAILED,
        payload: {
          cardId: card.id,
          issueKey,
          fromColumnId,
          toColumnId,
          reason: failure.reason,
          ...failure.extra,
        },
        errorMessage: failure.message,
        mappingId,
        reason: failure.reason,
      };
    }
  }

  private async resolveLinkedBoard(boardId: string): Promise<LinkedBoard> {
    try {
      return await this.statusMappings.requireLinkedBoard(boardId);
    } catch (error) {
      // Only the missing / not connected / not linked answers; a DB error stays `sync failed`.
      if (
        error instanceof NotFoundException ||
        error instanceof ConflictException ||
        error instanceof BadRequestException
      ) {
        throw new SyncFailure('board not linked', this.message(error));
      }
      throw error;
    }
  }

  private async getIssue(boardId: string, issueKey: string, deadline: number) {
    try {
      return await this.gateway.getIssue({
        boardId,
        issueIdOrKey: issueKey,
        timeBudgetMs: this.callBudget(deadline),
      });
    } catch (error) {
      if (error instanceof JiraClientError && error.status === 404) {
        throw new SyncFailure('issue not found', error.message);
      }
      throw error;
    }
  }

  private async listTransitions(
    boardId: string,
    issueKey: string,
    deadline: number,
  ): Promise<readonly JiraTransition[]> {
    const result = await this.gateway.listTransitions({
      boardId,
      issueIdOrKey: issueKey,
      includeUnavailableTransitions: false,
      timeBudgetMs: this.callBudget(deadline),
    });
    return result.transitions;
  }

  private async doTransition(
    boardId: string,
    issueKey: string,
    transitionId: string,
    deadline: number,
  ): Promise<void> {
    await this.gateway.doTransition({
      boardId,
      issueIdOrKey: issueKey,
      transitionId,
      timeBudgetMs: this.callBudget(deadline),
    });
  }

  private async findMappingId(boardId: string, issueKey: string): Promise<string | null> {
    const link = await this.links.findOne({
      where: { boardId, jiraIssueKey: issueKey },
      select: { id: true },
    });
    return link?.id ?? null;
  }

  private uniqueTargets(transitions: readonly JiraTransition[]): JiraStatus[] {
    const byId = new Map<string, JiraStatus>();
    for (const transition of transitions) {
      if (!byId.has(transition.to.id)) {
        byId.set(transition.to.id, transition.to);
      }
    }
    return [...byId.values()];
  }

  private compareCandidates(left: TransitionCandidate, right: TransitionCandidate): number {
    if (left.mapping.isDefault !== right.mapping.isDefault) {
      return left.mapping.isDefault ? 1 : -1;
    }
    return (
      this.compareIds(left.mapping.jiraStatusId, right.mapping.jiraStatusId) ||
      this.compareIds(left.transition.id, right.transition.id)
    );
  }

  private compareIds(left: string, right: string): number {
    if (/^\d+$/.test(left) && /^\d+$/.test(right)) {
      return Number(left) - Number(right);
    }
    return left.localeCompare(right);
  }

  private callBudget(deadline: number): number {
    const remaining = Math.min(deadline - Date.now(), SYNC_BUDGET_MS);
    if (remaining < MIN_CALL_BUDGET_MS) {
      throw new SyncFailure(
        'deadline exceeded',
        'The five-second synchronization budget is exhausted',
      );
    }
    return remaining;
  }

  private startedAt(occurredAt: string): number {
    const parsed = Date.parse(occurredAt);
    return Number.isNaN(parsed) ? Date.now() : parsed;
  }

  private failureFrom(error: unknown): {
    reason: FailureReason;
    message: string;
    extra?: Record<string, unknown>;
  } {
    if (error instanceof SyncFailure) {
      return { reason: error.reason, message: error.message, extra: error.extra };
    }
    if (error instanceof JiraTimeBudgetExceededError || error instanceof JiraTimeoutError) {
      return { reason: 'deadline exceeded', message: this.message(error) };
    }
    if (error instanceof JiraGatewayError) {
      return { reason: 'jira error', message: this.message(error) };
    }
    return { reason: 'sync failed', message: this.message(error) };
  }

  private message(error: unknown): string {
    return error instanceof Error ? error.message : String(error);
  }
}

import { Injectable, Logger } from '@nestjs/common';
import { OnEvent } from '@nestjs/event-emitter';
import { DomainEventOf } from '../../../common/events/domain-events';
import { JiraStatusSyncService } from './jira-status-sync.service';

/** Starts outbound status synchronization without delaying the card-move request. */
@Injectable()
export class JiraStatusSyncListener {
  private readonly logger = new Logger(JiraStatusSyncListener.name);

  constructor(private readonly sync: JiraStatusSyncService) {}

  @OnEvent('card.moved')
  onCardMoved(event: DomainEventOf<'card.moved'>): void {
    if (event.origin !== 'user') {
      return;
    }
    if (event.payload.fromColumnId === event.payload.toColumnId) {
      return;
    }
    if (event.payload.card.jiraIssueKey === null) {
      this.logger.debug(`Skipping Jira status sync for unlinked card ${event.payload.card.id}`);
      return;
    }

    void this.sync.syncCardMove(event).catch((error: unknown) => {
      const message = error instanceof Error ? error.message : String(error);
      const stack = error instanceof Error ? error.stack : undefined;
      this.logger.error(
        `Unexpected Jira status sync failure for card ${event.payload.card.id}: ${message}`,
        stack,
      );
    });
  }
}
